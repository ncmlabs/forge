FORGE Code Example

#! boundary: server

task classify_intent
  needs message: Text
  gives Classification
  do
    classify message into ["greeting", "question", "complaint"]

pure build_response
  needs intent: Classification
  gives Html
  do
    url = asset("css/style.css")
    match intent
      when "greeting"  give "<h1>Hello!</h1>"
      when "question"  give "<h1>Let me help.</h1>"
      when "complaint" give "<h1>I understand.</h1>"

endpoint respond(message: Text) -> Html
  intent = classify_intent(message)
  give build_response(intent)
